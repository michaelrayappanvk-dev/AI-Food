FOODAI

Your Customer Login OTP is:

{otp}

This OTP is valid for this login attempt.

Thank you for using FOODAI.

""" } try: response = requests.post( url, headers=headers, json=data, timeout=20 ) print("========== RESEND DEBUG ==========") print("RECIPIENT:", email) print("STATUS:", response.status_code) print("RESPONSE:", response.text) print("==================================") return response.status_code == 200 except Exception as e: print("RESEND ERROR:", str(e)) return False
